strategy MovingAverageCrossover {
symbol = "AAPL";
timeframe = {0,0,1};
p_size = 200;
indicators
{
    int a[2][2][2];
    timeseries b;
    a[b];
}
regimes {
rgm trending = atr14 > SMA(atr14, 20) * 1.5;
rgm ranging = atr14 <= SMA(atr14, 20) * 1.5;
}
trending::entry_long {
fast_ma > slow_ma AND
rsi_value > 40 AND
rsi_value < 75 &&
volume > volume_avg;
}
trending :: entry_short {
fast_ma < slow_ma AND
rsi_value > 25 AND
rsi_value < 60 AND
volume > volume_avg;
}
exit_long {
fast_ma < slow_ma OR
rsi_value > 80 OR
rsi_value < 25 OR
profit > 12.0 OR
loss > 6.0;
}
exit_short {
fast_ma > slow_ma OR
rsi_value < 20 OR
rsi_value > 75 OR
profit[0] > 10.0 OR
loss > 5.0;
}
}
fun name(
    int a, int b->int) {
    return a + b;
}
