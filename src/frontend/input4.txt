strategy MovingAverageCrossover 
{
    symbol = "AAPL";
    timeframe = {0,0,1};
    p_size = 200;

    indicators
    {
        timeseries fast_ma = SMA(close, 10);
        timeseries slow_ma = SMA(close, 20);
        timeseries volume_avg = SMA(volume, 15);
        timeseries atr14 = ATR(high, low, close, 14);
    }

    regimes 
    {
        rgm trending = atr14 > SMA(atr14, 20) * 1.5;
        rgm ranging = atr14 <= SMA(atr14, 20) * 1.5;
    }

    entry_long {
        fast_ma > slow_ma AND
        rsi_value > 40 AND
        rsi_value < 75 &&
        volume > volume_avg;
    }

    entry_short {
        fast_ma < slow_ma AND
        rsi_value > 25 AND
        rsi_value < 60 AND
        volume > volume_avg;
    }

    exit_long {
        fast_ma < slow_ma OR
        rsi_value > 80 OR
        rsi_value < 25 OR
        profit > 12.0 OR
        loss > 6.0;
    }

    exit_short {
        fast_ma > slow_ma OR
        rsi_value < 20 OR
        rsi_value > 75 OR
        profit[0] > 10.0 OR
        loss > 5.0;
    }
}

fun test( -> void)
{
    int n = 100;
    int steps = 0;
    int max_val = 0;

    while (n > 1) {
        if (n > max_val) {
            max_val = n;
        }

        if (n % 2 == 0) {
            n = n / 2;
        } else {
            n = (3 * n) + 1;
        }

        steps = steps + 1;
        
        if (steps > 1000) {
            return;
        }
    }
    return;
}
